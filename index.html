export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } if (request.method !== "POST") { return json({ ok: false, error: "Método no permitido" }, 405, corsHeaders); } const url = new URL(request.url); const body = await request.json().catch(() => null); if (!body || !body.playerId) { return json({ ok: false, error: "Solicitud incompleta" }, 400, corsHeaders); } // Recomendado: validar firma HMAC de body.initData contra env.BOT_TOKEN en producción const playerId = String(body.playerId).replace(/[^a-zA-Z0-9_-]/g, ""); const userKey = `user:${playerId}`; let user = await env.GAME_DB.get(userKey, { type: "json" }); if (!user) { user = { playerId, verifiedRealBalanceUsdt: 0, verifiedReferralCount: 0, verifiedFundedReferralCount: 0, verifiedWeeklyReferralCount: 0, redeemedReferralCode: null, influencerCommissionRate: 6.0, influencerEarnedUsdt: 0, influencerWithdrawalsCount: 0 }; } if (url.pathname === "/api/user/state") { await env.GAME_DB.put(userKey, JSON.stringify(user)); return json({ ok: true, ...user, weeklyLeaderboard: [] }, 200, corsHeaders); } // 1. Códigos de acreditación generados previamente por administración/soporte en KV if (url.pathname === "/api/credit/redeem") { const code = String(body.code || "").trim().toUpperCase(); if (!code) return json({ ok: false, error: "Código vacío" }, 400, corsHeaders); const codeKey = `credit_code:${code}`; const record = await env.GAME_DB.get(codeKey, { type: "json" }); if (!record || record.used) { return json({ ok: false, error: "Código inválido o ya utilizado" }, 400, corsHeaders); } if (record.assignedTo && String(record.assignedTo) !== playerId) { return json({ ok: false, error: "Este código pertenece a otra cuenta" }, 403, corsHeaders); } const amount = Number(record.amount) || 0; if (amount <= 0) { return json({ ok: false, error: "Monto de código inválido" }, 400, corsHeaders); } record.used = true; record.usedBy = playerId; record.usedAt = Date.now(); await env.GAME_DB.put(codeKey, JSON.stringify(record)); user.verifiedRealBalanceUsdt = Number(((Number(user.verifiedRealBalanceUsdt) || 0) + amount).toFixed(2)); await env.GAME_DB.put(userKey, JSON.stringify(user)); return json({ ok: true, creditedAmount: amount, verifiedRealBalanceUsdt: user.verifiedRealBalanceUsdt }, 200, corsHeaders); } // 2. Solicitud de retiro verificada contra saldo real del servidor if (url.pathname === "/api/withdraw/request") { const amount = Number(body.amount) || 0; const payoutAddress = String(body.payoutAddress || "").trim(); const payoutNetwork = String(body.payoutNetwork || "USDT-TON").trim(); if (amount < 1 || payoutAddress.length < 16) { return json({ ok: false, error: "Monto o billetera inválidos" }, 400, corsHeaders); } const currentBal = Number(user.verifiedRealBalanceUsdt) || 0; if (amount > currentBal) { return json({ ok: false, error: "Saldo real verificado insuficiente" }, 400, corsHeaders); } user.verifiedRealBalanceUsdt = Number((currentBal - amount).toFixed(2)); const withdrawId = `RET-${Date.now().toString().slice(-7)}`; await env.GAME_DB.put(userKey, JSON.stringify(user)); await env.GAME_DB.put(`withdraw:${withdrawId}`, JSON.stringify({ withdrawId, playerId, amount, payoutNetwork, payoutAddress, status: "pending_verification", createdAt: Date.now() })); return json({ ok: true, withdrawId, verifiedRealBalanceUsdt: user.verifiedRealBalanceUsdt }, 200, corsHeaders); } return json({ ok: false, error: "Ruta no encontrada" }, 404, corsHeaders); } }; function json(data, status = 200, headers = {}) { return new Response(JSON.stringify(data), { status, headers: { "Content-Type": "application/json", ...headers } }); }
